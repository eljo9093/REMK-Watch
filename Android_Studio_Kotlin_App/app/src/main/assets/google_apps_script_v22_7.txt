// - Unterstützt dein Tabellen-Layout:
//   Spalte A: Key/Passwort | Spalte B: Name | Spalte C: Status | Spalte D: Rolle
//   Spalte G: GERAET_A, B, C (Komma-getrennt) | Spalte H: Nickname | Spalte J: Wunsch-Passwort
//   (Erkennt auch automatisch, falls Spalte A = Name und Spalte B = Key ist!)
// - Admin / Trainer / TSV haben UNBEGRENZTE Geräte (niemals Geräte-Limit!)
// - Unterstützt den 1-Klick Kunden-Geräte-Reset (?action=resetDevices)
// ============================================================================

function getLicenseSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  return ss.getSheetByName("Lizenzen") || ss.getSheets()[0];
}

function jsonOut(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

// Erkennt automatisch anhand einer Tabellenzeile, ob Spalte A der Key ist (Standard) oder Spalte B
function parseRowColumns(row) {
  var colA = String(row[0] || "").trim();
  var colB = String(row[1] || "").trim();
  var colC = String(row[2] || "Aktiv").trim();
  var colD = String(row[3] || "Kunde").trim();
  var colE = String(row[4] || "").trim();
  var colF = String(row[5] || "").trim();
  var colG = String(row[6] || "").trim();
  var colH = String(row[7] || "").trim();
  var colI = row[8];
  var colJ = String(row[9] || "").trim();

  // Wenn in Spalte B ein typischer Name mit Leerzeichen steht (z.B. "Jochen Rube", "Tester 2 (Lisa)")
  // oder Spalte A Ziffern/Bindestriche wie "1", "26011990", "REMK-TEST-02" enthält -> Spalte A = Key, Spalte B = Name
  var colAIsKey = true;
  if (colA.indexOf(" ") !== -1 && colB.indexOf(" ") === -1 && colB !== "") {
    colAIsKey = false;
  }

  return {
    colAIsKey: colAIsKey,
    keyColOneBased: colAIsKey ? 1 : 2,
    nameColOneBased: colAIsKey ? 2 : 1,
    key: colAIsKey ? colA : colB,
    name: colAIsKey ? colB : colA,
    status: colC,
    role: colD,
    colE: colE,
    colF: colF,
    devicesRawG: colG,
    nickname: colH,
    expiryRaw: colI,
    customPwd: colJ
  };
}

function isUnlimitedAdminOrTrainer(roleStr, keyStr, nameStr) {
  var r = String(roleStr || "").toUpperCase();
  var k = String(keyStr || "").toUpperCase();
  var n = String(nameStr || "").toUpperCase();
  if (r.indexOf("ADMIN") !== -1 || r.indexOf("TRAINER") !== -1 || r.indexOf("TSV") !== -1 || r.indexOf("VIP") !== -1 || r.indexOf("COACH") !== -1) {
    return true;
  }
  if (k === "1" || k === "26011990" || k.indexOf("VIP") !== -1 || k.indexOf("TSV") !== -1 || n.indexOf("JOCHEN") !== -1) {
    return true;
  }
  return false;
}

function doGet(e) {
  try {
    var params = (e && e.parameter) ? e.parameter : {};
    var action = String(params.action || "verify").trim();
    var enteredUser = String(params.user || "").trim();
    var enteredKey = String(params.key || "").trim();
    var newNickname = String(params.nickname || "").trim();
    var newPassword = String(params.newKey || "").trim();
    var deviceId = String(params.deviceId || "WEB-DEVICE").trim();

    if (!enteredKey) {
      return jsonOut({ valid: false, message: "Bitte gib deinen Start-Key oder dein Passwort ein." });
    }

    var sheet = getLicenseSheet();
    var data = sheet.getDataRange().getValues();
    var MAX_DEVICES = 3;

    // ========================================================================
    // AKTION 1: GERÄTE ZURÜCKSETZEN (action === "resetDevices")
    // Leert alte Geräte in Spalte G (und ggf. E/F) und trägt das aktuelle Gerät ein!
    // ========================================================================
    if (action === "resetDevices") {
      for (var d = 1; d < data.length; d++) {
        var infoD = parseRowColumns(data[d]);
        if (!infoD.key && !infoD.customPwd) continue;

        var keyMatchD = (infoD.key !== "" && infoD.key.toLowerCase() === enteredKey.toLowerCase()) ||
                        (infoD.customPwd !== "" && infoD.customPwd === enteredKey);

        var userMatchD = !enteredUser ||
                         (infoD.name !== "" && infoD.name.toLowerCase().indexOf(enteredUser.toLowerCase()) !== -1) ||
                         (enteredUser.toLowerCase().indexOf(infoD.name.toLowerCase()) !== -1) ||
                         (infoD.nickname !== "" && infoD.nickname.toLowerCase() === enteredUser.toLowerCase()) ||
                         isUnlimitedAdminOrTrainer(infoD.role, infoD.key, infoD.name);

        if (keyMatchD && userMatchD) {
          // Setze Spalte G (Spalte 7) auf nur noch das aktuelle Gerät zurück!
          sheet.getRange(d + 1, 7).setValue(deviceId);
          // Falls in Spalte E/F alte DEV-IDs standen, ebenfalls bereinigen
          if (infoD.colE.indexOf("DEV-") === 0) sheet.getRange(d + 1, 5).setValue("");
          if (infoD.colF.indexOf("DEV-") === 0) sheet.getRange(d + 1, 6).setValue("");

          return jsonOut({
            valid: true,
            user: infoD.nickname || infoD.name || enteredUser || "Athlet",
            role: infoD.role || "Kunde",
            message: "✅ Alle alten Geräte wurden gelöscht und dieses Gerät neu aktiviert!"
          });
        }
      }
      return jsonOut({
        valid: false,
        message: "Konto für den Geräte-Reset nicht gefunden. Bitte prüfe Name/Nickname und Passwort/Key."
      });
    }

    // ========================================================================
    // AKTION 2: ERST-LOGIN NICKNAME FESTLEGEN (action === "setFirstNickname")
    // ========================================================================
    if (action === "setFirstNickname") {
      if (!newNickname || newNickname.length < 3) {
        return jsonOut({ valid: false, message: "Bitte wähle einen Nickname mit mindestens 3 Zeichen." });
      }
      for (var nIdx = 1; nIdx < data.length; nIdx++) {
        var infN = parseRowColumns(data[nIdx]);
        var sameRow = (infN.key.toLowerCase() === enteredKey.toLowerCase()) ||
                      (infN.customPwd !== "" && infN.customPwd === enteredKey);
        if (infN.nickname !== "" && infN.nickname.toLowerCase() === newNickname.toLowerCase() && !sameRow) {
          return jsonOut({ valid: false, message: "⚠️ Dieser Nickname ist bereits vergeben. Bitte wähle einen anderen!" });
        }
      }
      for (var r = 1; r < data.length; r++) {
        var infR = parseRowColumns(data[r]);
        if ((infR.key !== "" && infR.key.toLowerCase() === enteredKey.toLowerCase()) ||
            (infR.customPwd !== "" && infR.customPwd === enteredKey)) {
          sheet.getRange(r + 1, 8).setValue(newNickname); // Spalte H (8) = Nickname
          return jsonOut({
            valid: true,
            user: newNickname,
            role: infR.role || "Kunde",
            message: "✅ Nickname '" + newNickname + "' erfolgreich reserviert!"
          });
        }
      }
      return jsonOut({ valid: false, message: "Start-Key für Nickname-Vergabe nicht gefunden." });
    }

    // ========================================================================
    // AKTION 3: WUNSCH-PASSWORT ÄNDERN (action === "changePassword")
    // ========================================================================
    if (action === "changePassword") {
      if (!newPassword || newPassword.length < 4) {
        return jsonOut({ valid: false, message: "Das neue Passwort muss mindestens 4 Zeichen lang sein." });
      }
      for (var c = 1; c < data.length; c++) {
        var infC = parseRowColumns(data[c]);
        var kMatchC = (infC.key !== "" && infC.key.toLowerCase() === enteredKey.toLowerCase()) ||
                      (infC.customPwd !== "" && infC.customPwd === enteredKey);
        var uMatchC = !enteredUser ||
                      (infC.nickname !== "" && infC.nickname.toLowerCase() === enteredUser.toLowerCase()) ||
                      (infC.name !== "" && infC.name.toLowerCase().indexOf(enteredUser.toLowerCase()) !== -1);
        if (kMatchC && uMatchC) {
          sheet.getRange(c + 1, infC.keyColOneBased).setValue(newPassword);
          sheet.getRange(c + 1, 10).setValue(newPassword); // Zusätzlich in Spalte J sichern
          return jsonOut({
            valid: true,
            user: infC.nickname || infC.name || enteredUser,
            role: infC.role || "Kunde",
            message: "✅ Dein neues persönliches Passwort wurde erfolgreich gespeichert!"
          });
        }
      }
      return jsonOut({ valid: false, message: "Konto zum Ändern des Passworts nicht gefunden." });
    }

    // ========================================================================
    // AKTION 4: LOGIN- & LIZENZ-PRÜFUNG (action === "verify")
    // ========================================================================
    for (var i = 1; i < data.length; i++) {
      var info = parseRowColumns(data[i]);
      if (!info.key && !info.customPwd) continue;

      var keyMatches = (info.key !== "" && info.key.toLowerCase() === enteredKey.toLowerCase()) ||
                       (info.customPwd !== "" && info.customPwd === enteredKey);
      if (!keyMatches) continue;

      var isAdmin = isUnlimitedAdminOrTrainer(info.role, info.key, info.name);

      // 1. Status prüfen (Spalte C)
      var stLow = info.status.toLowerCase();
      if (stLow === "gesperrt" || stLow === "inaktiv" || stLow === "deaktiviert" || stLow === "abgelaufen") {
        return jsonOut({
          valid: false,
          locked: true,
          message: "🔒 Dieser Zugang ist aktuell pausiert oder gesperrt."
        });
      }

      // 2. Name / Nickname prüfen (Admin darf sich mit jedem Testnamen einloggen!)
      if (!isAdmin && enteredUser) {
        var nickSet = info.nickname !== "";
        var matchesNick = nickSet && info.nickname.toLowerCase() === enteredUser.toLowerCase();
        var matchesReal = info.name === "" ||
                          info.name.toLowerCase().indexOf("tester") !== -1 ||
                          info.name.toLowerCase() === enteredUser.toLowerCase() ||
                          info.name.toLowerCase().indexOf(enteredUser.toLowerCase()) !== -1 ||
                          enteredUser.toLowerCase().indexOf(info.name.split(" ")[0].toLowerCase()) !== -1;

        if (!matchesNick && !matchesReal) {
          return jsonOut({
            valid: false,
            message: nickSet
              ? ("⚠️ Bitte melde dich mit deinem Nickname (" + info.nickname + ") an.")
              : "⚠️ Name stimmt nicht mit dem Lizenz-Eintrag überein."
          });
        }
      }

      // 3. Geräte-Prüfung (Spalte G: Komma-getrennte Geräte-IDs)
      var devices = info.devicesRawG
        ? info.devicesRawG.split(",").map(function(x) { return x.trim(); }).filter(function(x) { return x.length > 0; })
        : [];

      if (deviceId && devices.indexOf(deviceId) === -1) {
        if (isAdmin) {
          // ADMIN / TRAINER FREIFAHRTSCHEIN: Niemals blockieren! Behalte die letzten 3 IDs in Spalte G bei
          devices.push(deviceId);
          if (devices.length > MAX_DEVICES) {
            devices = devices.slice(devices.length - MAX_DEVICES);
          }
          sheet.getRange(i + 1, 7).setValue(devices.join(", "));
        } else if (devices.length < MAX_DEVICES) {
          devices.push(deviceId);
          sheet.getRange(i + 1, 7).setValue(devices.join(", "));
        } else {
          return jsonOut({
            valid: false,
            locked: true,
            message: "🚫 Geräte-Limit erreicht (max. 3 Geräte aktiv). Klicke unten auf 'Alte Geräte zurücksetzen'!"
          });
        }
      }

      var needsFirstSetup = (!isAdmin && info.nickname === "");

      return jsonOut({
        valid: true,
        user: info.nickname || info.name || enteredUser || "Athlet",
        realName: info.name || enteredUser || "Athlet",
        role: info.role || (isAdmin ? "Admin" : "Kunde"),
        needsFirstSetup: needsFirstSetup,
        unlimitedDevices: isAdmin
      });
    }

    return jsonOut({
      valid: false,
      message: "❌ Ungültiger Start-Key oder falsches Passwort. Bitte prüfe deine Eingabe."
    });

  } catch (err) {
    return jsonOut({ valid: false, message: "Serverfehler: " + err.toString() });
  }
}
